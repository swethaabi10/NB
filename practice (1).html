from sklearn.tree import DecisionTreeClassifier

# Features:
# [age, months_knowing, daily_chat_minutes, weekly_meet_count]
X = [
    [18, 1, 10, 0],   # just started, small chats
    [19, 2, 15, 0],
    [20, 3, 20, 1],   # mild crush
    [21, 4, 30, 1],
    [22, 6, 45, 2],   # strong like
    [23, 8, 60, 2],
    [24, 10, 80, 3],  # love
    [25, 12, 90, 3],
    [26, 15, 120, 4], # deep love
    [27, 18, 150, 4]
]

# Labels:
# 0 = "just friend", 1 = "crush", 2 = "like", 3 = "love", 4 = "deep love"
y = [0, 0, 1, 1, 2, 2, 3, 3, 4, 4]

DT = DecisionTreeClassifier(random_state=42)
DT.fit(X, y)

def explain_level(label):
    mapping = {
        0: "just friend",
        1: "crush",
        2: "like",
        3: "love",
        4: "deep love"
    }
    return mapping[int(label)]

# Try some scenarios
test_data = [
    [20, 2, 15, 0],   # early chats
    [23, 7, 50, 2],   # more time together
    [26, 16, 140, 4],# full-on relationship
    [24,36,180,6] 
]

preds = DT.predict(test_data)

for features, pred in zip(test_data, preds):
    print(features, "->", explain_level(pred))

[23, 7, 50, 2] -> like
[26, 16, 140, 4] -> deep love
